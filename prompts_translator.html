// ─── MYMEMORY TRANSLATE (works from file://) ────────────────────────────────── async function gtTranslate(text, targetLang) { // MyMemory free API — no CORS issues, works from file:// // Free limit: 500 chars/req, 1000 req/day (no key) // For longer texts we split by \n and rejoin const MAX_CHUNK = 480; if (text.length <= MAX_CHUNK) { return await mmTranslate(text, targetLang); } // Split long text into lines, batch into chunks ≤480 chars const lines = text.split('\n'); const chunks = []; let cur = ''; for (const line of lines) { const sep = cur ? '\n' : ''; if ((cur + sep + line).length <= MAX_CHUNK) { cur += sep + line; } else { if (cur) chunks.push(cur); // If single line > MAX_CHUNK, split by sentence if (line.length > MAX_CHUNK) { const sentences = line.match(/.{1,480}/g) || [line]; for (const s of sentences) chunks.push(s); cur = ''; } else { cur = line; } } } if (cur) chunks.push(cur); const results = []; for (const chunk of chunks) { const t = await mmTranslate(chunk, targetLang); results.push(t); await sleep(100); // small delay between chunks } return results.join('\n'); } async function mmTranslate(text, targetLang) { // Normalize lang code for MyMemory (zh-CN → zh-CN, iw → he, etc.) let mmLang = targetLang; if (mmLang === 'iw') mmLang = 'he'; const url = `https://api.mymemory.translated.net/get?q=${encodeURIComponent(text)}&langpair=cs|${mmLang}`; // Retry logic for rate limiting - much longer delays let lastError; for (let attempt = 0; attempt < 3; attempt++) { try { const resp = await fetch(url); if (!resp.ok) { if (resp.status === 429) { // Rate limited - wait much longer and retry await sleep(5000 * (attempt + 1)); // 5s, 10s, 15s lastError = new Error('MyMemory error ' + resp.status); continue; } throw new Error('MyMemory error ' + resp.status); } const json = await resp.json(); if (json.responseStatus !== 200) throw new Error('MyMemory: ' + json.responseMessage); return json.responseData.translatedText; } catch (error) { lastError = error; if (attempt < 2) { // Don't sleep on last attempt await sleep(3000 * (attempt + 1)); // 3s, 6s } } } throw lastError; } // Apply regular substitutions (without the CZ fix) function applyRegularSubstitutions(text, langCode) { // Sort longest search first const sorted = [...substitutions].sort((a,b) => b.search.length - a.search.length); for (const sub of sorted) { const replacement = sub.replaces[langCode]; if (!replacement) continue; // Case-insensitive replacement const escaped = sub.search.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); text = text.replace(new RegExp(escaped, 'gi'), replacement); } return text; } // Apply substitutions after translation (includes regular subs and CZ fix) function applySubstitutions(text, langCode) { // Apply regular substitutions text = applyRegularSubstitutions(text, langCode); // Replace standalone "CZ" with target language code (uppercase) // This fixes consistency rules like "nach CZ" -> "nach DE" for German translation text = text.replace(/\bCZ\b/g, targetLang.toUpperCase()); return text; } // Also apply substitutions to the CS source before translation // so translator doesn't get confused by "czechtina" phrases function prepareForTranslation(text, langCode) { // First apply regular subs to source (without the CZ replacement) return applyRegularSubstitutions(text, langCode); } // ─── TRANSLATE ONE ───────────────────────────────────────────────────────────── async function translateOne(langCode) { if (!csData) { toast('Nejprve načti JSON soubor', 'err'); return; } const lang = LANGUAGES.find(l => l.code === langCode);
